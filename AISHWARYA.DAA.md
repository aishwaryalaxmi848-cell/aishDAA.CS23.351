## Program 8 - Search for a given element in an array using Linear Search

## Aim

To write a Python program to search for a given element in an array using Linear Search.

## Program

```python
n = int(input("Enter the size of the array: "))

arr = []

for i in range(n):
    element = int(input("Enter element: "))
    arr.append(element)

key = int(input("Enter the element to search: "))

found = False
i = 0

while i < n and not found:
    if arr[i] == key:
        found = True
    else:
        i = i + 1

if found:
    print("Element found at position", i + 1)
else:
    print("Element not found")
```

```text
Function Main
    Declare Integer n
    Declare Integer i
    Declare Integer key
    Declare Boolean found

    Input n
    Declare Integer Array arr[n]

    For i = 0 to n - 1
        Input arr[i]
    End For

    Output "Enter the element to search:"
    Input key

    Assign found = False
    Assign i = 0
    While i < n AND found = False

            Assign i = i + 1
```

## Program 13 - Fibonacci Series using Recursion

## Aim
Write a python program to generate Fibonacci series using recursion

## Program
```python
def fibonacci(n):
    # Base cases
    if n <= 0:
        return 0
    elif n == 1:
        return 1
    # Recursive case
    else:
        return fibonacci(n - 1) + fibonacci(n - 2)
```

```python
def insertion_sort(arr):
    for i in range(1, len(arr)):
        key = arr[i]
        j = i - 1

        # Move elements of arr[0..i-1] that are greater than key
        # to one position ahead of their current position
        while j >= 0 and key < arr[j]:
            arr[j + 1] = arr[j]
            j -= 1
        arr[j + 1] = key
```

```python
from collections import deque

class Node:
    def __init__(self, data):
        self.data = data
        self.neighbors = []

class Graph:
    def __init__(self):
        self.nodes = []

    def add_node(self, data):
        node = Node(data)
        self.nodes.append(node)
        return node

    def add_edge(self, node1, node2):
        node1.neighbors.append(node2)
        node2.neighbors.append(node1)

    def print_graph(self):
        for node in self.nodes:
            print(node.data, "->", end=" ")
            for neighbor in node.neighbors:
                print(neighbor.data, end=" ")
            print()

    def BFS(self, start):
        visited = set()
        queue = deque([start])
        visited.add(start)

        while queue:
            node = queue.popleft()
            print(node.data, end=" ")

            for neighbor in node.neighbors:
                if neighbor not in visited:
                    visited.add(neighbor)
                    queue.append(neighbor)
```

## Program 35 - Dijkstra Algorithm

## Aim
Write a python program to show Dijkstra Algorithm

## Program
```python
class Node:
    def __init__(self, data):
        self.data = data
        self.neighbors = []

class Graph:
    def __init__(self):
        self.nodes = []

    def add_node(self, data):
        node = Node(data)
        self.nodes.append(node)
        return node

    def add_edge(self, node1, node2, weight):
        node1.neighbors.append((node2, weight))
        node2.neighbors.append((node1, weight))

    def print_graph(self):
        for node in self.nodes:
            print(node.data, "->", end=" ")

            for neighbor, weight in node.neighbors:
                print(f"{neighbor.data}({weight})", end=" ")

            print()

    def dijkstra(self, source):
        distance = {}

        # Step 1: Initialize distances
        for node in self.nodes:
            distance[node] = float('inf')

        distance[source] = 0

        visited = set()

        # Step 2: Repeat until all nodes are visited
        while len(visited) < len(self.nodes):

            # Find minimum-distance unvisited node
            current = None
            min_distance = float('inf')

            for node in self.nodes:
                if node not in visited and distance[node] < min_distance:
                    min_distance = distance[node]
                    current = node

            # No reachable node left
            if current is None:
                break

            # Mark current node visited
            visited.add(current)

            # Relax edges
            for neighbor, weight in current.neighbors:
                new_distance = distance[current] + weight

                if new_distance < distance[neighbor]:
                    distance[neighbor] = new_distance

        return distance

# Create graph
g = Graph()

A = g.add_node("A")
B = g.add_node("B")
C = g.add_node("C")
D = g.add_node("D")

# Add weighted edges
g.add_edge(A, B, 4)
g.add_edge(A, C, 2)
g.add_edge(C, B, 1)
g.add_edge(B, D, 3)
g.add_edge(C, D, 5)

# Find shortest distances from A
distance = g.dijkstra(A)

# Print result
for node, dist in distance.items():
    print(node.data, "=", dist)

```

## Program 36 - Bell-Man Ford Algorithm

## Aim
Write a python program to demonstrate Bell-Man Ford algorithm.

## Program
```python
class Node:
    def __init__(self, data):
        self.data = data
        self.neighbors = []

class Graph:
    def __init__(self):
        self.nodes = []

    def add_node(self, data):
        node = Node(data)
        self.nodes.append(node)
        return node

    def add_edge(self, node1, node2, weight):
        node1.neighbors.append((node2, weight))
        #node2.neighbors.append((node1, weight))

    def print_graph(self):
        for node in self.nodes:
            print(node.data, "->", end=" ")

            for neighbor, weight in node.neighbors:
                print(f"{neighbor.data}({weight})", end=" ")

            print()

    def bellman_ford(self, source):

        # Step 1: Initialize distances
        distance = {}

        for node in self.nodes:
            distance[node] = float('inf')

        distance[source] = 0

        # Step 2: Relax all edges V-1 times
        for i in range(len(self.nodes) - 1):

            for node in self.nodes:

                for neighbor, weight in node.neighbors:

                    if distance[node] != float('inf'):

                        new_distance = distance[node] + weight

                        if new_distance < distance[neighbor]:
                            distance[neighbor] = new_distance

        # Step 3: Check for negative weight cycle
        for node in self.nodes:

            for neighbor, weight in node.neighbors:

                if distance[node] != float('inf'):

                    if distance[node] + weight < distance[neighbor]:
                        print("Negative weight cycle detected!")
                        return None

        return distance
```
